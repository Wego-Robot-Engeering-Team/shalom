cmake_minimum_required(VERSION 3.16)
project(slamtec_aurora)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)

set(AURORA_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party/aurora_ros/src/aurora_remote_public")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  set(AURORA_SDK_ARCH linux_aarch64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64)$")
  set(AURORA_SDK_ARCH linux_x86_64)
else()
  message(FATAL_ERROR "Unsupported Aurora SDK architecture: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
set(AURORA_SDK_LIBRARY "${AURORA_SDK_ROOT}/lib/${AURORA_SDK_ARCH}/libslamtec_aurora_remote_sdk.so")
if(NOT EXISTS "${AURORA_SDK_LIBRARY}")
  message(FATAL_ERROR "Aurora SDK library not found: ${AURORA_SDK_LIBRARY}")
endif()

add_executable(aurora_person_cloud src/person_cloud_node.cpp)
target_compile_features(aurora_person_cloud PRIVATE cxx_std_17)
target_include_directories(aurora_person_cloud PRIVATE "${AURORA_SDK_ROOT}/include" ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(aurora_person_cloud rclcpp sensor_msgs)
target_link_libraries(aurora_person_cloud "${AURORA_SDK_LIBRARY}" ${OpenCV_LIBRARIES})
install(TARGETS aurora_person_cloud DESTINATION lib/${PROJECT_NAME})

install(DIRECTORY config launch rviz DESTINATION share/${PROJECT_NAME})

ament_package()
